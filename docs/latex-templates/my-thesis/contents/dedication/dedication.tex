\begin{center}
	{\itshape\large
	Tuhan menciptakan langit dan bumi dalam ketetapan-Nya.\\
	Ia menjadikan siang dan malam sebagai tanda bagi yang berpikir.\\
	Ia membentuk setiap pribadi dengan maksud yang mulia, \\
	menempa hamba-Nya menjadi pejuang yang tangguh,\\
	pembawa terang di tengah gelap, saksi kebenaran di muka bumi.\\
	Sungguh, segala yang dijadikan-Nya amat baik.}
	\vspace{2cm}
\end{center}

Skripsi ini kupersembahkan kepada:\vspace{0.5cm}

\begin{enumerate}[leftmargin=*]
	\item \textbf{Bunda, Maria Magdalena}, ibu terkasih yang tiada henti berjuang seorang diri mengusahakan yang terbaik untuk ketiga anaknya.
	
	\item \textbf{Syifa Aulia}, kekasih yang menemani perjuangan akademik dan profesional, yang terus meyakinkan bahwa semua bisa diselesaikan dengan sempurna.
	
	\item \textbf{Hazon El Yeshua Manalu dan Waldensia Ebenz Shiloh Manalu}, adik-adik terkasih yang meski tak sering berjumpa, menjadi alasan untuk bisa menjadi teladan yang baik.
	
	\item \textbf{Teteh Siti Rena Haryani}, yang tak henti mengatakan hidup selalu bisa dihadapi, tak peduli pahitnya masa lalu.
	
	\item \textbf{Father, Yand Abraham Manalu}, ayah yang mendukung sebagian masa kecil, melandasi jalan kehidupan berikutnya.
	
	\item \textbf{Keluarga Manalu}, yang selalu berharap dan mendoakan pendidikan terbaik.
	
	\item \textbf{Seluruh rekan-rekan kuliah dari bumi Solo dan Yogyakarta}, yang mendukung perjalanan hidup kemahasiswaan.
\end{enumerate}
