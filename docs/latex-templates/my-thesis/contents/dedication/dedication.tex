\begin{center}
	{\itshape\large
	Tuhan menciptakan langit dan bumi dalam ketetapan-Nya.\\
	Ia menjadikan siang dan malam sebagai tanda bagi yang berpikir.\\
	Ia membentuk setiap pribadi dengan maksud yang mulia---\\
	menempa hamba-Nya menjadi pejuang yang tangguh,\\
	pembawa terang di tengah gelap, saksi kebenaran di muka bumi.\\
	Sungguh, segala yang dijadikan-Nya amat baik.}
	\vspace{2cm}
\end{center}

Skripsi ini kupersembahkan kepada:\vspace{0.5cm}

\begin{enumerate}[leftmargin=*]
	\item \textbf{Bunda, Maria Magdalena}---ibu tercinta yang seorang diri memperjuangkan masa depan ketiga anaknya. Tiada satu hari pun terlewat tanpa doa dan jerih payahmu.
	
	\item \textbf{Syifa Aulia}---yang setia mendampingi dan mendukung sepenuhnya selama proses akademik maupun non-akademik.
	
	\item \textbf{Hazon El Yeshua Manalu dan Waldensia Ebenz Shiloh Manalu}---adik-adik terkasih yang menjadi alasan untuk terus menjadi teladan.
	
	\item \textbf{Teteh Rena Aryani}---yang tak lelah mengingatkan bahwa hidup selalu bisa dihadapi dan disyukuri.
	
	\item \textbf{Father, Yand Abraham Delomo Manalu}---ayah yang membekali dasar berpikir dan semangat juang.
	
	\item \textbf{Keluarga besar Manalu}---yang terus menaruh harapan dan mendoakan pendidikan terbaik.
	
	\item \textbf{Rekan-rekan kuliah di bumi Solo dan Yogyakarta}---yang mewarnai perjalanan kemahasiswaan, mendukung di setiap langkah, dan menjadi saksi perjuangan ini.
\end{enumerate}
